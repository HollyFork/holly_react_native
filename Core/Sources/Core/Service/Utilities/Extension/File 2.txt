extension PlanningDto {
    func toDomain() -> Planning {
        Planning(
            restaurant: Restaurant(
                id: restaurant.id,
                nom: restaurant.nom
            ),
            semaine: Semaine(
                debut: semaine.debut,
                fin: semaine.fin
            ),
            jours: jours.map {
                Jour(
                    date: $0.date,
                    jour: $0.jour,
                    creneaux: $0.creneaux.map {
                        Creneau(debut: $0.debut, fin: $0.fin)
                    },
                    totalHeures: $0.totalHeures
                )
            },
            totalSemaineHeures: totalSemaineHeures
        )
    }
}
